HERE = Path(__file__).resolve().parent
REPO = HERE.parents[1]
MANIFEST = json.loads((HERE / "manifest.json").read_text(encoding="utf-8"))
TEXT_EXTENSIONS = {".md", ".txt", ".json", ".csv", ".py", ".js", ".yaml", ".yml"}
PUBLIC_ITEM_FIELDS = {
    "agent_message": {"id", "type", "text"},
    "command_execution": {"id", "type", "command", "aggregated_output", "exit_code", "status"},
    "file_change": {"id", "type", "changes", "status"},
    "mcp_tool_call": {"id", "type", "server", "tool", "arguments", "result", "error", "status"},
    "web_search": {"id", "type", "query", "action", "status"},
    "todo_list": {"id", "type", "items"},
    "error": {"id", "type", "message"},
}


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def canonical(value) -> bytes:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode("utf-8")


def write_json(path: Path, value) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def under(root: Path, target: Path) -> None:
    if not target.resolve().is_relative_to(root.resolve()):
        raise ValueError("Path escapes the intended root")


def safe_member(name: str) -> bool:
    path = PurePosixPath(name)
    return not path.is_absolute() and ".." not in path.parts and "\\" not in name and ":" not in name


def runtime_bytes(revision: str | None = None) -> dict[str, bytes]:
    if revision:
        archive = subprocess.run(["git", "archive", "--format=tar", revision, *MANIFEST["runtime_paths"]],
                                 cwd=REPO, check=True, capture_output=True).stdout
        result = {}
        with tarfile.open(fileobj=io.BytesIO(archive)) as stream:
            for member in stream.getmembers():
                if not safe_member(member.name) or member.issym() or member.islnk():
                    raise ValueError("Unsafe Git archive member")
                if member.isfile():
                    result[member.name] = stream.extractfile(member).read()
        return result
    result = {}
    for relative in MANIFEST["runtime_paths"]:
        entry = REPO / relative
        for path in ([entry] if entry.is_file() else sorted(entry.rglob("*"))):
            if path.is_file() and "__pycache__" not in path.parts and path.suffix != ".pyc":
                if path.is_symlink():
                    raise ValueError("Runtime symlinks are not supported")
                result[path.relative_to(REPO).as_posix()] = path.read_bytes()
    return result


def hashes(files: dict[str, bytes]) -> dict[str, str]:
    return {name: sha(data) for name, data in sorted(files.items())}


def tree_hashes(root: Path) -> dict[str, str]:
    return hashes({p.relative_to(root).as_posix(): p.read_bytes() for p in regular_files(root)})


def regular_files(root: Path) -> list[Path]:
    """Do not silently treat an inaccessible sandbox directory as an empty directory."""
    output = []

    def fail(error):
        raise error

    for current, directories, names in os.walk(root, onerror=fail, followlinks=False):
        directories[:] = [name for name in directories if not (Path(current) / name).is_symlink()]
        for name in names:
            path = Path(current) / name
            if not path.is_symlink():
                output.append(path)
    return sorted(output)


def prepare_host_read(project: Path) -> dict:
    """Give the host read access to its own temporary outputs before the sandbox writes."""
    under(Path(tempfile.gettempdir()), project)
    if project.name != "project" or not re.fullmatch(r"elenchus-research-eval-[A-Za-z0-9_-]+", project.parent.name):
        raise ValueError("Host read setup is limited to exact runner-created temporary project roots")
    if os.name != "nt":
        return {"status": "not_applicable"}
    started = datetime.now(timezone.utc).isoformat()
    identity = subprocess.run(["whoami", "/user", "/fo", "csv", "/nh"], capture_output=True, check=True)
    match = re.search(r"S-1-5-[0-9-]+", identity.stdout.decode("ascii", errors="replace"))
    if not match:
        raise RuntimeError("Could not resolve the current host SID for temporary output reading")
    grant = subprocess.run(["icacls", str(project), "/grant", f"*{match.group(0)}:(OI)(CI)R"],
                           capture_output=True, check=False)
    if grant.returncode:
        raise RuntimeError(f"Temporary host read setup failed with exit code {grant.returncode}")
    return {"status": "host_read_ace_prepared", "scope": "temporary project root, inheritable host read only",
            "before_model_start": True, "started_utc": started,
            "finished_utc": datetime.now(timezone.utc).isoformat(), "exit_code": grant.returncode}


def cleanup_temporary(temporary: Path) -> dict:
    under(Path(tempfile.gettempdir()), temporary)
    if not temporary.name.startswith("elenchus-research-eval-"):
        raise ValueError("Cleanup is limited to verified evaluation temporary roots")
    started = datetime.now(timezone.utc).isoformat()
    try:
        shutil.rmtree(temporary)
        return {"status": "complete", "started_utc": started, "finished_utc": datetime.now(timezone.utc).isoformat()}
    except OSError as error:
        return {"status": "retained_on_error", "started_utc": started,
                "finished_utc": datetime.now(timezone.utc).isoformat(), "error": str(error),
                "retained_temp": "<os-temp>/" + temporary.name}


def freeze(variant: str, label: str | None) -> Path:
    head = subprocess.run(["git", "rev-parse", "HEAD"], cwd=REPO, check=True,
                          capture_output=True, encoding="utf-8", errors="replace").stdout.strip()
    revision = MANIFEST["baseline_revision"] if variant == "baseline" else head
    source = runtime_bytes(revision)
    if variant == "candidate":
        normalized_source = hashes({key: data.replace(b"\r\n", b"\n") for key, data in source.items()})
        normalized_worktree = hashes({key: data.replace(b"\r\n", b"\n") for key, data in runtime_bytes().items()})
        if normalized_source != normalized_worktree:
            raise ValueError("Candidate runtime must be committed before freeze")
    files = hashes(source)
    name = label or variant
    if not re.fullmatch(r"[a-zA-Z0-9_-]+", name):
        raise ValueError("Snapshot label must be a plain identifier")
    destination = HERE / "snapshots" / f"{name}.json"
    fixture_hashes = tree_hashes(HERE / "fixtures")
    value = {"schema_version": 1, "variant": variant, "label": name,
             "revision": revision, "files": files, "source_sha256": sha(canonical(files)),
             "head_revision": head,
             "normalized_lf_files": hashes({key: data.replace(b"\r\n", b"\n") for key, data in source.items()}),
             "manifest_sha256": sha((HERE / "manifest.json").read_bytes()),
             "fixture_hashes": fixture_hashes, "fixtures_sha256": sha(canonical(fixture_hashes))}
    if destination.exists():
        if json.loads(destination.read_text(encoding="utf-8")) != value:
            raise FileExistsError("Snapshot exists with different bytes; choose a new label")
    else:
        write_json(destination, value)
    return destination


def verify_snapshot(snapshot: dict) -> dict[str, bytes]:
    source = runtime_bytes(snapshot.get("revision"))
    if hashes(source) != snapshot["files"]:
        raise ValueError("Runtime changed after freeze; freeze a new candidate label")
    if sha((HERE / "manifest.json").read_bytes()) != snapshot["manifest_sha256"]:
        raise ValueError("Manifest changed after freeze")
    if tree_hashes(HERE / "fixtures") != snapshot["fixture_hashes"]:
        raise ValueError("Fixture bytes changed after freeze")
    return source


class Redactor:
    def __init__(self, replacements: dict[str, str] | None = None):
        self.replacements = replacements or {}

    def text(self, value: str) -> str:
        for old, replacement in sorted(self.replacements.items(), key=lambda item: -len(item[0])):
            value = re.sub(re.escape(old), lambda _: replacement, value, flags=re.I)
            value = re.sub(re.escape(old.replace("\\", "/")), lambda _: replacement, value, flags=re.I)
        value = re.sub(r"(?i)[A-Z]:[\\/]+Users[\\/]+[^\\/\s\"<>]+", "<user>", value)
        value = re.sub(r"(?i)(Bearer\s+)[A-Za-z0-9._~+/=-]+", r"\1<redacted>", value)
        value = re.sub(r"\bsk-[A-Za-z0-9_-]{12,}\b", "<redacted-key>", value)
        value = re.sub(r"\beyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\b", "<redacted-token>", value)
        value = re.sub(r'(?i)((?:access_token|refresh_token|id_token|api_key|authorization)[\"\s:=]+)[^\s,}\"]+',
                       r"\1<redacted>", value)
        return value

    def apply(self, value):
        if isinstance(value, str):
            return self.text(value)
        if isinstance(value, list):
            return [self.apply(item) for item in value]
        if isinstance(value, dict):
            return {key: ("<redacted>" if key.lower() in
                          {"access_token", "refresh_token", "id_token", "api_key", "authorization"}
                          else self.apply(item)) for key, item in value.items()}
        return value


def public_events(raw: str, redactor: Redactor) -> tuple[list[dict], dict[str, int]]:
    public, excluded = [], {}
    for line in raw.splitlines():
        try:
            event = json.loads(line)
        except (json.JSONDecodeError, TypeError):
            excluded["non_json"] = excluded.get("non_json", 0) + 1
            continue
        kind = event.get("type", "unknown")
        if kind.startswith("item."):
            item = event.get("item", {})
            item_kind = item.get("type", "unknown")
            fields = PUBLIC_ITEM_FIELDS.get(item_kind)
            if not fields:
                excluded[item_kind] = excluded.get(item_kind, 0) + 1
                continue
            item = {key: value for key, value in item.items() if key in fields}
            command = str(item.get("command", ""))
            if re.search(r"(?i)\.codex[\\/](?:auth\.json|config\.toml)", command):
                item = {"id": item.get("id"), "type": item_kind,
                        "privacy_redacted": True, "status": item.get("status")}
            public.append(redactor.apply({"type": kind, "item": item}))
        elif kind in {"thread.started", "turn.started", "turn.completed", "turn.failed", "error"}:
            fields = {"type", "thread_id", "usage", "error", "message"}
            public.append(redactor.apply({key: value for key, value in event.items() if key in fields}))
        else:
            excluded[kind] = excluded.get(kind, 0) + 1
    return public, excluded


def cli_command(cli: str, project: Path, output: Path, model: str, effort: str) -> list[str]:
    cmd = [cli, "-a", "never", "exec", "--ignore-user-config", "--ignore-rules", "--ephemeral",
           "--skip-git-repo-check", "--sandbox", "workspace-write", "-c", 'web_search="disabled"',
           "-c", "features.multi_agent=false", "-m", model, "-c", f'model_reasoning_effort="{effort}"',
           "-C", str(project), "--json", "-o", str(output)]
    if os.name == "nt":
        cmd += ["-c", 'windows.sandbox="elevated"']
    skill_root = Path.home() / ".codex" / "skills"
    disabled = [str(path).replace("\\", "/") for path in skill_root.glob("elenchus*/SKILL.md")]
    if disabled:
        rows = ",".join('{path="' + path.replace('"', '\\"') + '",enabled=false}' for path in disabled)
        cmd += ["-c", "skills.config=[" + rows + "]"]
    return cmd + ["-"]


def call_cli(command: list[str], prompt: str, timeout: float) -> dict:
    started = time.monotonic()
    flags = subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0
    environment = os.environ.copy()
    environment.update({"PYTHONUTF8": "1", "PYTHONIOENCODING": "utf-8"})
    if os.name == "nt":
        import ctypes
        ctypes.windll.kernel32.SetConsoleCP(65001)
        ctypes.windll.kernel32.SetConsoleOutputCP(65001)
    try:
        process = subprocess.Popen(command, stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                                   encoding="utf-8", errors="replace", creationflags=flags, env=environment)
    except OSError as error:
        return {"stdout": "", "stderr": str(error), "exit_code": None, "timed_out": False,
                "startup_error": True, "elapsed_seconds": round(time.monotonic() - started, 3)}
    timed_out = False
    try:
        stdout, stderr = process.communicate(prompt, timeout=max(0.1, timeout))
    except subprocess.TimeoutExpired:
        timed_out = True
        if os.name == "nt":
            subprocess.run(["taskkill", "/PID", str(process.pid), "/T", "/F"], capture_output=True,
                           creationflags=flags, check=False)
        if process.poll() is None:
            process.kill()
        stdout, stderr = process.communicate()
    return {"stdout": stdout, "stderr": stderr, "exit_code": process.returncode,
            "timed_out": timed_out, "elapsed_seconds": round(time.monotonic() - started, 3)}

def prompt_for(history: list[tuple[str, str]]) -> str:
    if len(history) == 1:
        return history[0][1]
    previous = "\n\n".join(f"### {role}\n{text}" for role, text in history[:-1])
    return ("아래는 이 프로젝트의 같은 계획 대화에서 이미 공개된 이전 대화입니다. 다른 평가 결과가 아닙니다.\n\n"
            + previous + "\n\n### 현재 사용자 응답\n" + history[-1][1])


def reported_verdict(project: Path, answer: str) -> bool:
    notes = project / ".elenchus"
    if not notes.exists():
        return False
    documents = [path for path in regular_files(notes) if path.parent.name == "research" and
                 path.name.startswith("R") and path.suffix == ".md"]
    if not documents:
        return False
    for path in documents:
        content = path.read_text(encoding="utf-8", errors="replace")
        if ("Verdict" not in content and "판정" not in content) or not re.search(r"결론 가능|결론 보류", content):
            return False
        if re.search(r"상태\W{0,8}확정", content):
            continue
        if "결론 보류" in content and ("보류" in answer or "공백" in answer):
            continue
        return False
    return True


def collect_artifacts(project: Path, output: Path, redactor: Redactor) -> dict:
    artifacts = {}
    notes = project / ".elenchus"
    if not notes.exists():
        return artifacts
    for path in regular_files(notes):
        relative = path.relative_to(project).as_posix()
        raw = path.read_bytes()
        artifacts[relative] = {"sha256": sha(raw), "bytes": len(raw), "published": False}
        if path.suffix in TEXT_EXTENSIONS and len(raw) <= 262144:
            destination = output / "artifacts" / relative
            under(output, destination)
            destination.parent.mkdir(parents=True, exist_ok=True)
            published = redactor.text(raw.decode("utf-8", errors="replace")).encode("utf-8")
            destination.write_bytes(published)
            artifacts[relative].update({"published": True, "published_sha256": sha(published)})
    return artifacts


def run_sample(args) -> Path:
    snapshot = json.loads(args.snapshot.read_text(encoding="utf-8"))
    source = verify_snapshot(snapshot)
    output = HERE / "results" / f'{args.label or snapshot["label"]}-{args.case}-sample{args.sample}'
    output.mkdir(parents=True, exist_ok=False)
    cli_version = subprocess.run([args.cli, "--version"], capture_output=True, encoding="utf-8", errors="replace", check=False).stdout.strip()
    summary = {"kind": "controlled", "status": "running", "variant": snapshot["variant"],
               "case": args.case, "sample": args.sample, "snapshot": snapshot,
               "cli_version": cli_version, "model": MANIFEST["model"],
               "reasoning_effort": MANIFEST["reasoning_effort"],
               "runner_sha256": sha(Path(__file__).read_bytes()),
               "protocol_sha256": sha((HERE / "README.md").read_bytes()),
               "started_utc": datetime.now(timezone.utc).isoformat(),
               "limits": {"max_responses": MANIFEST["controlled"]["max_responses"],
                          "max_seconds": MANIFEST["controlled"]["max_seconds"]},
               "turns": [], "quality_verdict": "not_automatically_graded"}
    write_json(output / "summary.json", summary)
    case_root = HERE / "fixtures" / args.case
    history = [("user", (case_root / "request.md").read_text(encoding="utf-8"))]
    temp_root = Path(tempfile.mkdtemp(prefix="elenchus-research-eval-")).resolve()
    under(Path(tempfile.gettempdir()), temp_root)
    redactor = Redactor({str(temp_root): "<run-temp>", str(REPO): "<repository>", str(Path.home()): "<user>"})
    try:
        project = temp_root / "project"
        project.mkdir()
        summary["host_read_setup"] = prepare_host_read(project)
        summary["temporary_workspace"] = "<os-temp>/" + temp_root.name
        prepare_project(project, source, args.case)
        before = tree_hashes(project)
        redactor = Redactor({str(project): "<fixture>", str(temp_root): "<run-temp>",
                             str(REPO): "<repository>", str(Path.home()): "<user>"})
        started = time.monotonic()
        status = "response_limit"
        for turn_number in range(1, MANIFEST["controlled"]["max_responses"] + 1):
            remaining = MANIFEST["controlled"]["max_seconds"] - (time.monotonic() - started)
            if remaining <= 0:
                status = "timeout"
                break
            turn_dir = output / f"turn-{turn_number:02d}"
            turn_dir.mkdir()
            prompt = prompt_for(history)
            published_prompt = redactor.text(prompt)
            (turn_dir / "input.md").write_text(published_prompt, encoding="utf-8")
            final_path = temp_root / f"final-{turn_number}.md"
            command = cli_command(args.cli, project, final_path, MANIFEST["model"], MANIFEST["reasoning_effort"])
            result = call_cli(command, prompt, remaining)
            events, excluded = public_events(result.pop("stdout"), redactor)
            stderr = redactor.text(result.pop("stderr"))
            (turn_dir / "events.jsonl").write_text("".join(json.dumps(e, ensure_ascii=False) + "\n" for e in events), encoding="utf-8")
            (turn_dir / "stderr.txt").write_text(stderr, encoding="utf-8")
            answer = final_path.read_text(encoding="utf-8", errors="replace") if final_path.exists() else ""
            if not answer:
                visible = [e["item"].get("text", "") for e in events if e.get("type") == "item.completed"
                           and e.get("item", {}).get("type") == "agent_message"]
                answer = visible[-1] if visible else ""
            published_answer = redactor.text(answer)
            (turn_dir / "answer.md").write_text(published_answer, encoding="utf-8")
            usage = [e["usage"] for e in events if e.get("type") == "turn.completed" and "usage" in e]
            tool_calls = [e for e in events if e.get("type") == "item.completed" and
                          e.get("item", {}).get("type") in {"command_execution", "mcp_tool_call", "web_search", "file_change"}]
            turn = {"number": turn_number, **result, "input_sha256": sha(prompt.encode("utf-8")),
                    "published_input_sha256": sha(published_prompt.encode("utf-8")),
                    "answer_sha256": sha(answer.encode("utf-8")),
                    "published_answer_sha256": sha(published_answer.encode("utf-8")),
                    "excluded_event_types": excluded, "usage": usage, "tool_calls": len(tool_calls),
                    "question_line_candidates": sum(bool(re.search(r"[?？]|까요|인가요", line)) for line in answer.splitlines()),
                    "command": redactor.apply(command)}
            summary["turns"].append(turn)
            summary["elapsed_seconds"] = round(time.monotonic() - started, 3)
            write_json(output / "summary.json", summary)
            print(json.dumps({"run": output.name, "turn": turn_number, "elapsed_seconds": summary["elapsed_seconds"],
                              "exit_code": result["exit_code"], "timed_out": result["timed_out"]}), flush=True)
            if result["timed_out"]:
                status = "timeout"
                break
            if result["exit_code"] != 0 or any(e.get("type") == "turn.failed" for e in events):
                status = "cli_error"
                break
            if not answer:
                status = "empty_response"
                break
            history.append(("assistant", answer))
            try:
                if reported_verdict(project, answer):
                    status = "reported_research_verdict"
                    break
            except OSError as error:
                summary.setdefault("collection_warnings", []).append(
                    {"turn": turn_number, "error": redactor.text(str(error))})
            next_input = (case_root / "preferences.md" if turn_number == 1 else HERE / "fixtures" / "continuation.md")
            history.append(("user", next_input.read_text(encoding="utf-8")))
        try:
            after = tree_hashes(project)
            changes = [{"path": name, "before": before.get(name), "after": after.get(name)}
                       for name in sorted(before.keys() | after.keys()) if before.get(name) != after.get(name)]
            summary.update({"artifacts": collect_artifacts(project, output, redactor), "artifact_collection": "complete",
                            "file_changes": changes, "product_unchanged": before["app.txt"] == after.get("app.txt"),
                            "out_of_plan_changes": [change for change in changes if not change["path"].startswith(".elenchus/")]})
        except OSError as error:
            summary.update({"artifact_collection": "incomplete", "artifacts": None, "file_changes": None,
                            "product_unchanged": None, "out_of_plan_changes": None,
                            "collection_error": redactor.text(str(error)),
                            "retained_temp": "<os-temp>/" + temp_root.name})
        summary.update({"status": status, "elapsed_seconds": round(time.monotonic() - started, 3),
                        "finished_utc": datetime.now(timezone.utc).isoformat()})
        write_json(output / "summary.json", summary)
    except BaseException as error:
        summary["host_infrastructure_error"] = redactor.text(str(error))
        if summary["status"] == "running":
            summary["status"] = "infrastructure_error"
        raise
    finally:
        summary["cleanup"] = redactor.apply(cleanup_temporary(temp_root))
        summary["cleanup_status"] = summary["cleanup"]["status"]
        if temp_root.exists():
            summary["retained_temp"] = "<os-temp>/" + temp_root.name
        write_json(output / "summary.json", summary)
    return output


def probe(args) -> Path:
    if args.timeout_check:
        return probe_timeout()
    if args.tools:
        return probe_tools(args)
    output = HERE / "results" / f"setup-{datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%S')}"
    output.mkdir(parents=True, exist_ok=False)
    project = Path(tempfile.mkdtemp(prefix="elenchus-research-eval-basic-probe-")).resolve()
    under(Path(tempfile.gettempdir()), project)
    redactor = Redactor({str(project): "<probe>", str(Path.home()): "<user>"})
    record = {"kind": "setup", "status": "running"}
    try:
        final_path = project / "answer.md"
        command = cli_command(args.cli, project, final_path, MANIFEST["model"], MANIFEST["reasoning_effort"])
        result = call_cli(command, "Respond exactly READY. Do not use tools or read any files.", 120)
        events, excluded = public_events(result.pop("stdout"), redactor)
        result["stderr"] = redactor.text(result["stderr"])
        record.update({**result, "status": "response_observed" if result["exit_code"] == 0 else "cli_error", "events": events,
                       "excluded_event_types": excluded, "command": redactor.apply(command)})
    except BaseException as error:
        record.update({"status": "infrastructure_error", "host_infrastructure_error": redactor.text(str(error))})
        raise
    finally:
        record["cleanup"] = redactor.apply(cleanup_temporary(project))
        write_json(output / "summary.json", record)
    return output


def probe_tools(args) -> Path:
    output = HERE / "results" / f"setup-tools-{datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%S')}"
    output.mkdir(parents=True, exist_ok=False)
    temporary = Path(tempfile.mkdtemp(prefix="elenchus-research-eval-setup-")).resolve()
    under(Path(tempfile.gettempdir()), temporary)
    project = temporary / "project"
    project.mkdir()
    redactor = Redactor({str(project): "<fixture>", str(temporary): "<probe-temp>", str(Path.home()): "<user>"})
    setup = prepare_host_read(project)
    expected = "한글 읽기·쓰기 검증 — 카지노 사진 정리 123\n문서 상태: 작성 중\n"
    (project / "fixture-ko.txt").write_text(expected, encoding="utf-8", newline="")
    prompt = ("이것은 실행 환경 준비 점검입니다. 사용자 스킬·인증·설정·네트워크는 읽지 마세요. "
              "평소 사용하는 셸 도구로 fixture-ko.txt를 읽어 도구 출력에 내용을 보이세요. "
              "그다음 셸 도구로 .elenchus/research/ 디렉터리를 만들고 같은 내용을 "
              ".elenchus/research/probe-ko.txt에 복사한 뒤 셸 도구로 그 파일을 다시 읽어 출력하세요. "
              "두 번 읽은 내용이 같은지 확인하고 마지막 답변에는 읽은 텍스트를 그대로 한 번만 적으세요.")
    (output / "input.md").write_text(prompt, encoding="utf-8")
    final_path = temporary / "final.md"
    command = cli_command(args.cli, project, final_path, MANIFEST["model"], MANIFEST["reasoning_effort"])
    result = call_cli(command, prompt, 180)
    events, excluded = public_events(result.pop("stdout"), redactor)
    result["stderr"] = redactor.text(result["stderr"])
    event_text = "".join(json.dumps(event, ensure_ascii=False) + "\n" for event in events)
    (output / "events.jsonl").write_text(event_text, encoding="utf-8")
    answer = final_path.read_text(encoding="utf-8", errors="replace") if final_path.exists() else ""
    (output / "answer.md").write_text(redactor.text(answer), encoding="utf-8")
    completed_commands = [event["item"] for event in events if event.get("type") == "item.completed"
                          and event.get("item", {}).get("type") == "command_execution"]
    tool_outputs = "\n".join(item.get("aggregated_output", "") for item in completed_commands)
    checks = {"model_completed": result["exit_code"] == 0 and not result["timed_out"],
              "actual_shell_commands": len(completed_commands),
              "korean_in_tool_output": expected.splitlines()[0] in tool_outputs,
              "no_replacement_characters": "\ufffd" not in tool_outputs,
              "korean_in_final_answer": expected.splitlines()[0] in answer}
    record = {"kind": "setup", "mode": "korean_tools_and_artifact_roundtrip", **result,
              "runner_sha256": sha(Path(__file__).read_bytes()), "host_read_setup": setup,
              "explicit_encoding": "UTF-8 stdin/stdout; Python UTF8/IO env; Windows console CP 65001",
              "input_sha256": sha(prompt.encode("utf-8")), "fixture_sha256": sha(expected.encode("utf-8")),
              "events_sha256": sha(event_text.encode("utf-8")), "excluded_event_types": excluded,
              "usage": [event["usage"] for event in events if event.get("type") == "turn.completed" and "usage" in event],
              "command": redactor.apply(command), "checks": checks,
              "temporary_workspace": "<os-temp>/" + temporary.name}
    try:
        record["files"] = tree_hashes(project)
        actual = (project / ".elenchus/research/probe-ko.txt").read_bytes()
        checks["host_can_read_new_artifact"] = True
        checks["artifact_bytes_match"] = actual == expected.encode("utf-8")
        record["artifacts"] = collect_artifacts(project, output, redactor)
    except OSError as error:
        checks["host_can_read_new_artifact"] = False
        checks["artifact_bytes_match"] = None
        record["collection_error"] = redactor.text(str(error))
    record["setup_passed"] = all(value is True for key, value in checks.items() if key != "actual_shell_commands") and len(completed_commands) > 0
    write_json(output / "summary.json", record)
    record["cleanup"] = redactor.apply(cleanup_temporary(temporary))
    write_json(output / "summary.json", record)
    return output


def probe_timeout() -> Path:
    """A real bounded child process, with no model call or fabricated model event trace."""
    output = HERE / "results" / f"setup-timeout-{datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%S')}"
    output.mkdir(parents=True, exist_ok=False)
    temporary = Path(tempfile.mkdtemp(prefix="elenchus-research-eval-timeout-")).resolve()
    under(Path(tempfile.gettempdir()), temporary)
    artifact = temporary / "before-timeout.txt"
    script = ("import pathlib,sys,time; pathlib.Path(sys.argv[1]).write_text('BEFORE_TIMEOUT',encoding='utf-8'); "
              "print('TIMEOUT_PROBE_STARTED',flush=True); time.sleep(20)")
    result = call_cli([sys.executable, "-u", "-c", script, str(artifact)], "", 1)
    public_stdout = result.pop("stdout")
    (output / "stdout.txt").write_text(public_stdout, encoding="utf-8")
    redactor = Redactor({str(temporary): "<probe-temp>", str(Path.home()): "<user>"})
    checks = {"timed_out": result["timed_out"], "prefix_output_preserved": "TIMEOUT_PROBE_STARTED" in public_stdout,
              "artifact_preserved": artifact.exists() and artifact.read_text(encoding="utf-8") == "BEFORE_TIMEOUT",
              "returned_within_five_seconds": result["elapsed_seconds"] < 5}
    if artifact.exists():
        (output / "before-timeout.txt").write_bytes(artifact.read_bytes())
    record = {"kind": "setup", "mode": "bounded_process_timeout", "model_invoked": False,
              **redactor.apply(result), "deadline_seconds": 1, "checks": checks, "setup_passed": all(checks.values()),
              "runner_sha256": sha(Path(__file__).read_bytes())}
    write_json(output / "summary.json", record)
    record["cleanup"] = redactor.apply(cleanup_temporary(temporary))
    write_json(output / "summary.json", record)
    return output


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    sub = parser.add_subparsers(dest="command", required=True)
    freeze_parser = sub.add_parser("freeze", help="Freeze immutable runtime and fixture hashes")
    freeze_parser.add_argument("--variant", choices=("baseline", "candidate"), required=True)
    freeze_parser.add_argument("--label")
    probe_parser = sub.add_parser("probe", help="Separate CLI/auth setup probe, not a measured sample")
    probe_parser.add_argument("--cli", default="codex")
    probe_parser.add_argument("--tools", action="store_true", help="Exercise Korean shell I/O and host export of sandbox-created artifacts")
    probe_parser.add_argument("--timeout-check", action="store_true", help="Verify interrupted child-process output and artifact preservation; no model call")
    run_parser = sub.add_parser("run", help="Run one independent controlled sample")
    run_parser.add_argument("--snapshot", type=Path, required=True)
    run_parser.add_argument("--case", choices=MANIFEST["controlled"]["cases"], required=True)
    run_parser.add_argument("--sample", type=int, choices=(1, 2), required=True)
    run_parser.add_argument("--cli", default="codex")
    run_parser.add_argument("--label", help="New result label for explicitly separated retests")
    args = parser.parse_args()
    if args.command == "run" and args.label and not re.fullmatch(r"[a-zA-Z0-9_-]+", args.label):
        raise ValueError("Use a plain result label identifier")
    if args.command == "run" and (HERE / "MEASURED_RUNS_PAUSED").exists():
        snapshot = json.loads(args.snapshot.read_text(encoding="utf-8"))
        paused_name = f'paused-{args.label or snapshot["label"]}-{args.case}-sample{args.sample}'
        destination = HERE / "results" / paused_name
        destination.mkdir(parents=True, exist_ok=False)
        write_json(destination / "summary.json", {"kind": "controlled", "status": "not_run_infrastructure_pause",
                   "variant": snapshot["variant"], "case": args.case, "sample": args.sample,
                   "source_sha256": snapshot["source_sha256"], "model_invoked": False,
                   "recorded_utc": datetime.now(timezone.utc).isoformat()})
        text = {"status": "not_run_infrastructure_pause", "result": str(destination)}
        print(json.dumps(text), flush=True)
        return
    destination = freeze(args.variant, args.label) if args.command == "freeze" else probe(args) if args.command == "probe" else run_sample(args)
    print(str(destination), flush=True)


if __name__ == "__main__":
    main()
